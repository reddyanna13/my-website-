create table if not exists public.user_accounts ( id bigint generated by default as identity primary key, user_id uuid not null references auth.users(id) on delete cascade, label text not null, created_at timestamptz not null default now() ); create table if not exists public.sales_history ( id bigint generated by default as identity primary key, user_id uuid not null references auth.users(id) on delete cascade, description text not null, amount numeric(12,2) not null check (amount > 0), created_at timestamptz not null default now() ); alter table public.user_accounts enable row level security; alter table public.sales_history enable row level security; create policy "Read own accounts" on public.user_accounts for select to authenticated using ((select auth.uid()) = user_id); create policy "Insert own accounts" on public.user_accounts for insert to authenticated with check ((select auth.uid()) = user_id); create policy "Read own sales" on public.sales_history for select to authenticated using ((select auth.uid()) = user_id); create policy "Insert own sales" on public.sales_history for insert to authenticated with check ((select auth.uid()) = user_id);
